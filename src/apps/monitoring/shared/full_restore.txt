import React, { useState } from 'react';
import { Route as DashRoute } from './routes/dashboard';
import { Route as MapRoute } from './routes/index';
import { Route as DataRoute } from './routes/data';
import { Route as ReportsRoute } from './routes/reports';
import { Route as AIRoute } from './routes/ai-assistant';
import './styles.css';

export default function Monitoring({ onBack }) {
  const [activeTab, setActiveTab] = useState('/');

  const renderContent = () => {
    switch (activeTab) {
      case '/': return <MapRoute.options.component />;
      case '/dashboard': return <div className='h-full overflow-y-auto'><DashRoute.options.component /></div>;
      case '/data': return <div className='h-full overflow-y-auto'><DataRoute.options.component /></div>;
      case '/reports': return <div className='h-full overflow-y-auto'><ReportsRoute.options.component /></div>;
      case '/ai-assistant': return <AIRoute.options.component />;
      default: return <MapRoute.options.component />;
    }
  };

  return (
    <div className='monitoring-theme-ffb h-screen w-full overflow-hidden bg-background text-foreground'>
      <AppShellOverride activeTab={activeTab} setActiveTab={setActiveTab} onBack={onBack}>
        {renderContent()}
      </AppShellOverride>
    </div>
  );
}

import { LayoutDashboard, Map, FileText, Bot, Droplets, Bell, Search, User, Database, LogOut, Trees } from "lucide-react";

function AppShellOverride({ activeTab, setActiveTab, onBack, children }) {
  const nav = [
    { to: "/", label: "Estate Map", icon: Map },
    { to: "/dashboard", label: "Operations", icon: LayoutDashboard },
    { to: "/reports", label: "Production", icon: FileText },
    { to: "/data", label: "Field Data", icon: Database },
    { to: "/ai-assistant", label: "Assistant", icon: Bot },
  ];

  return (
    <div className="flex h-full bg-background">
      <aside className="hidden md:flex w-64 flex-col border-r bg-sidebar">
        <div className="px-6 py-6 flex items-center gap-3 border-b">
          <div className="h-10 w-10 rounded-xl bg-primary text-primary-foreground grid place-items-center shadow-lg">
            <Trees className="h-6 w-6" />
          </div>
          <div>
            <p className="text-sm font-bold tracking-tight">PalmSense Pro</p>
            <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Industrial Monitoring</p>
          </div>
        </div>
        <nav className="flex-1 px-4 py-6 space-y-1.5">
          {nav.map((n) => (
            <button key={n.to} onClick={() => setActiveTab(n.to)}
              className={`flex w-full items-center gap-3 px-4 py-2.5 rounded-lg text-sm font-medium transition ${
                activeTab === n.to ? "bg-primary text-primary-foreground shadow-lg" : "text-muted-foreground hover:bg-accent/40 hover:text-foreground"
              }`}>
              <n.icon className="h-4 w-4" />
              {n.label}
            </button>
          ))}
        </nav>
        <div className="p-4 border-t">
          <button onClick={onBack} className="flex w-full items-center gap-2 px-4 py-2 text-xs text-muted-foreground hover:text-foreground transition mb-4">
             <LogOut className="h-4 w-4" /> Back to Hub
          </button>
          <div className="p-3 rounded-xl bg-accent/20 border border-border/50">
            <div className="flex items-center gap-2 text-xs font-semibold text-primary mb-1">
              <Droplets className="h-3 w-3" /> Soil Moisture
            </div>
            <p className="text-lg font-bold">82% <span className="text-[10px] font-normal text-muted-foreground">Optimal</span></p>
          </div>
        </div>
      </aside>

      <div className="flex-1 flex flex-col min-w-0">
        <header className="h-16 border-b bg-card/50 backdrop-blur flex items-center px-6 gap-4 sticky top-0 z-20">
          <div className="relative flex-1 max-w-md">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
            <input placeholder="Search estate, divisions, batches…" className="w-full h-10 pl-10 pr-4 text-sm bg-accent/20 border-transparent rounded-xl focus:outline-none focus:ring-2 focus:ring-primary/20" />
          </div>
          <div className="flex items-center gap-3 ml-auto">
            <button className="h-10 w-10 flex items-center justify-center hover:bg-accent rounded-xl border border-border/50"><Bell className="h-4 w-4" /></button>
            <button className="h-10 w-10 flex items-center justify-center hover:bg-accent rounded-xl border border-border/50"><User className="h-4 w-4" /></button>
          </div>
        </header>
        <main className="flex-1 overflow-hidden relative">{children}</main>
      </div>
    </div>
  );
}
